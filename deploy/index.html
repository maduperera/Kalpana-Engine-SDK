Python Quick start example 
import requests

# 1. Ingest a document
with open("report.pdf", "rb") as f:
    response = requests.post(
        "https://madurox-kalpana-api-public"
".hf.space/ingest",
        files={"file": f}
    )

# 2. Query memory
answer = requests.post(
    "https://madurox-kalpana-api-public"
".hf.space/chat",
    json={
        "message": "What were the key findings?",
        "session_id": response.json()["session_id"]
    }
)

print(answer.json()["response"])
# → "The key findings were..."
